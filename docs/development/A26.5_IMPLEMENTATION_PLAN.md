## 8. Presentation mapper contract (`src/features/availability/availabilityPresentation.ts`)

Pure function; imports only frozen domain types + `getRegionName`. Does NOT modify A26.2.

```ts
import type {
  AvailabilityResult,
  AvailabilityType,
  ProviderErrorKind,
} from "../../domain/availability/types";
import {
  getRegionName,
  type AvailabilityRegionCode,
} from "../settings/region/availabilityRegion";

/** Presentation-only mirror of the domain's private AVAILABILITY_TYPE_ORDER
 *  (types.ts ~L302). A26.2 is not modified to export it. */
const UI_AVAILABILITY_TYPE_ORDER: readonly AvailabilityType[] = [
  "flatrate", "free", "ads", "rent", "buy",
] as const;

const ACCESS_GROUP_LABELS: Record<AvailabilityType, string> = {
  flatrate: "Subscription",
  free: "Free",
  ads: "With ads",
  rent: "Rent",
  buy: "Buy",
};

const ERROR_COPY: Record<ProviderErrorKind, string> = {
  rateLimited: "Too many requests. Try again later.",
  serviceUnavailable: "The availability service is temporarily unavailable.",
  timeout: "The availability check timed out.",
  network: "Network error while checking availability.",
  unknown: "Could not check availability right now.",
  other: "Could not check availability right now.",
};

const ERROR_KIND_PRIORITY: readonly ProviderErrorKind[] = [
  "rateLimited", "serviceUnavailable", "timeout", "network", "unknown", "other",
];

export type AvailabilityViewState =
  | "available" | "unavailable" | "partial" | "error" | "offline";

export interface AvailabilityGroupView {
  readonly type: AvailabilityType;
  readonly label: string;
}

export interface AvailabilityViewModel {
  readonly state: AvailabilityViewState;
  readonly heading: string;                // "Where to watch (India)"
  readonly region: AvailabilityRegionCode; // verbatim from result.region
  readonly groups: readonly AvailabilityGroupView[];
  readonly link?: string;                  // verbatim from result.link, or omitted
  readonly statusMessage?: string;
  readonly isRetryable: boolean;           // true ONLY for state "error"
}

export function toAvailabilityViewModel(
  result: AvailabilityResult,
): AvailabilityViewModel {
  const regionName = getRegionName(result.region);
  const heading = `Where to watch (${regionName})`;

  // Defensive: out-of-catalogue group types are dropped, never rendered.
  const groups = result.groups
    .filter((g) => (UI_AVAILABILITY_TYPE_ORDER as readonly string[]).includes(g.type))
    .map((g) => ({
      type: g.type as AvailabilityType,
      label: ACCESS_GROUP_LABELS[g.type as AvailabilityType],
    }))
    .sort((a, b) =>
      UI_AVAILABILITY_TYPE_ORDER.indexOf(a.type) -
      UI_AVAILABILITY_TYPE_ORDER.indexOf(b.type));

  const { verdict, resolution } = result;
  let state: AvailabilityViewState;
  let statusMessage: string | undefined;
  let isRetryable = false;

  switch (resolution) {
    case "offline":
      state = "offline";
      statusMessage = "You are offline. Availability cannot be checked right now.";
      break;
    case "error":
      state = "error";
      isRetryable = true;
      statusMessage = dominantErrorCopy(result);
      break;
    case "partial":
      state = "partial";
      statusMessage = "Some availability information couldn't be checked.";
      break;
    case "resolved":
      if (verdict === "available") {
        state = "available"; // degenerate 0-groups case renders heading/link only
      } else if (verdict === "unavailable") {
        state = "unavailable";
        statusMessage = `Not available in ${regionName}.`;
      } else {
        state = "unavailable"; // verdict "unknown"
        statusMessage =
          `No availability information is available for this title in ${regionName}.`;
      }
      break;
  }

  return {
    state,
    heading,
    region: result.region,
    groups,
    link: result.link,
    statusMessage,
    isRetryable,
  };
}

function dominantErrorCopy(result: AvailabilityResult): string {
  const kinds = new Set(
    result.providerResults
      .map((r) => r.error?.kind)
      .filter((k): k is ProviderErrorKind => k !== undefined),
  );
  for (const kind of ERROR_KIND_PRIORITY) {
    if (kinds.has(kind)) {
      return ERROR_COPY[kind];
    }
  }
  return ERROR_COPY.unknown;
}
```

## 9. React fetch lifecycle (`src/features/availability/MediaAvailabilitySection.tsx`)

```tsx
import { useEffect, useState } from "react";
import { LoaderCircle } from "lucide-react";
// Same Media type the A26.4 service accepts (exact module path pinned at S1).
import type { Media } from "../../types/media";
import type { AvailabilityResult } from "../../domain/availability/types";

import { getAvailabilityForMedia } from "../../services/availability/mediaAvailabilityService";
import { useOnlineStatus } from "../../app/useOnlineStatus";
import { useRegion } from "../settings/region/regionContext";
import { getRegionName } from "../settings/region/availabilityRegion";

import { toAvailabilityViewModel } from "./availabilityPresentation";

export interface MediaAvailabilitySectionProps {
  media: Media;
}

export default function MediaAvailabilitySection({ media }: MediaAvailabilitySectionProps) {
  const { region, status, openRegionSelection } = useRegion();
  const isOnline = useOnlineStatus();

  const [result, setResult] = useState<AvailabilityResult | null>(null);
  const [loadError, setLoadError] = useState<Error | null>(null);
  const [attempt, setAttempt] = useState(0);

  // D2: region must be confirmed before any availability request.
  const isRegionReady = status === "ready";

  useEffect(() => {
    if (!isRegionReady) {
      setResult(null);
      setLoadError(null);
      return; // no getAvailabilityForMedia(); no IN fallback
    }
    let cancelled = false;
    setResult(null);
    setLoadError(null);

    void getAvailabilityForMedia(media, region, { isOnline: () => isOnline })
      .then((next) => {
        if (!cancelled) {
          setResult(next);
          setLoadError(null);
        }
      })
      .catch((error: unknown) => {
        if (!cancelled) {
          // Invalid-region rejection is unreachable; defensive only (D2).
          setLoadError(
            error instanceof Error ? error : new Error("Unable to check availability."),
          );
          setResult(null);
        }
      });

    return () => {
      cancelled = true; // stale in-flight results never overwrite newer state
    };
  }, [media.id, media.mediaType, media.tmdbId, region, status, isOnline, attempt]);

  const CTA_CLASS =
    "mt-3 inline-flex items-center gap-2 rounded-lg bg-accent px-4 py-2 text-sm font-medium text-inverted transition hover:bg-accent-hover focus:outline-none focus-visible:ring-2 focus-visible:ring-accent-hover/40";

  if (!isRegionReady) {
    return (
      <section aria-label="Where to watch">
        <h2 className="text-2xl font-semibold text-primary">Where to watch</h2>
        <p className="mt-3 text-sm text-muted">
          Set your region to check where you can watch this title.
        </p>
        <button type="button" onClick={openRegionSelection} className={CTA_CLASS}>
          Choose region
        </button>
      </section>
    );
  }

  if (loadError !== null) {
    return (
      <section aria-label="Where to watch">
        <h2 className="text-2xl font-semibold text-primary">
          Where to watch ({getRegionName(region)})
        </h2>
        <p
          role="status"
          className="mt-3 rounded-lg border border-danger/60 bg-danger/10 px-4 py-3 text-sm text-danger"
        >
          Could not check availability right now.
        </p>
        <button type="button" onClick={() => setAttempt((c) => c + 1)} className={CTA_CLASS}>
          Try again
        </button>
      </section>
    );
  }

  if (result === null) {
    return (
      <section
        aria-label="Where to watch"
        aria-busy="true"
        className="mt-8 rounded-xl border border-border bg-surface p-6"
      >
        <h2 className="text-2xl font-semibold text-primary">Where to watch</h2>
        <div className="mt-3 flex items-center gap-2 text-sm text-muted">
          <LoaderCircle className="h-4 w-4 animate-spin" aria-hidden="true" />
          <span>Checking availability…</span>
        </div>
      </section>
    );
  }

  const viewModel = toAvailabilityViewModel(result);
  return (
    <AvailabilitySectionView
      viewModel={viewModel}
      onRetry={() => setAttempt((c) => c + 1)}
    />
  );
}
```

`AvailabilitySectionView` (same file) renders strictly from `AvailabilityViewModel`: `h2` = `viewModel.heading`; group badges from `viewModel.groups` (text labels — never color-only); `<p role="status">` for `viewModel.statusMessage` when present; external link iff `viewModel.link` is defined — `<a href={…} target="_blank" rel="noopener noreferrer" aria-label="View watch options in {getRegionName(viewModel.region)}">`; "Try again" `<button>` iff `viewModel.isRetryable` (D6). No direct domain access inside the view.

## 13. Performance considerations

- No new dependency (`react`, `lucide-react`, Tailwind already present); no new route or chunk.
- One availability request per meaningful change: effect dependencies are primitives (`media.id`, `media.mediaType`, `media.tmdbId`, `region`, `status`, `isOnline`, `attempt`), so unrelated re-renders (e.g., an edit-save mutating `details`) do NOT refetch.
- No polling, caching, persistence, startup fetch, or bulk queries.
- The mapper is synchronous and trivial.
- Region-context init transient (recorded, NOT fixed per instructions): `RegionProvider` initializes to the default region with `status: "ready"` before the async preference load resolves, so the component may issue at most one fetch during that transient — using the context's current region (never a hard-coded fallback), self-correcting when `status` changes, with stale results cancelled. No region-preference architecture change (D2).

## 14. Risks and mitigations

- Pre-existing untracked ledger with false baseline claims → MITIGATED: superseded by this verified ledger; baseline re-checked independently in this gate (§2, §19).
- Region-context init transient (default region / `ready` pre-load) → at most one corrective fetch; never a silent `IN` fallback; deferred, not fixed (per instructions).

- Going offline during an in-flight request → resolves as a normal error, then the `isOnline` dependency refetches once on reconnect → converges offline → resolved.
- No DOM/component tests → mapper coverage must be comprehensive (D3); the component stays thin.
- Stale result on media/region change mid-flight → `cancelled` flag (same pattern as `MovieDetailsPage`'s details load) prevents overwrite.
